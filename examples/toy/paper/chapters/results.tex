\begin{lemma}\label{lem:succ}
  For every $n$, $f(n) > n$. % a comment is not part of the statement
\end{lemma}
\begin{lemma}[Banach]\label{lem:banach}
  Every complete normed space satisfies the Banach property.
\end{lemma}
\begin{theorem}[Twice]\label{thm:twice}
  Under the Banach property,
  \[ f(f(n)) > n. \]
\end{theorem}
\begin{equation}\label{eq:f}
  f(n) = n + 1
\end{equation}
